def compute_zscore(series, window=20):
    rm = series.rolling(window).mean()
    rs = series.rolling(window).std()
    return (series - rm) / rs

df["z_score"] = compute_zscore(df["mid_price"])
df["z_sig"] = (df["z_score"].abs() > 2.4).astype(int)
